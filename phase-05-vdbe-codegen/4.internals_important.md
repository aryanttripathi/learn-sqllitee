## 2. Structures

### 2.1 `VdbeOp` — one instruction

```c
struct VdbeOp {
  u8 opcode;
  signed char p4type;   /* one of the P4_* constants */
  u16 p5;               /* OPFLAG_* flags */
  int p1, p2, p3;
  union p4union {
    int i; void *p; char *z; i64 *pI64; double *pReal;
    FuncDef *pFunc; sqlite3_context *pCtx; CollSeq *pColl;
    Mem *pMem; sqlite3_vtab *pVtab; KeyInfo *pKeyInfo; u32 *ai;
    SubProgram *pProgram; Table *pTab; ...
  } p4;
#ifdef SQLITE_ENABLE_EXPLAIN_COMMENTS
  char *zComment;
#endif
#ifdef SQLITE_ENABLE_STMT_SCANSTATUS
  u32 nExec; u64 nCycle;
#endif
};
```

Operand conventions (consistent across all ~190 opcodes):

```
P1  usually a CURSOR number; sometimes a register
P2  usually a JUMP TARGET;   sometimes a register
P3  usually an OUTPUT register
P4  typed extra data — see the P4_* table
P5  flags — see the OPFLAG_* table
```

### 2.2 `P4_*` types

```c
#define P4_NOTUSED       0    /* not used */
#define P4_TRANSIENT     0    /* a transient string */
#define P4_STATIC      (-1)   /* a static string */
#define P4_COLLSEQ     (-2)   /* a CollSeq* */
#define P4_INT32       (-3)
#define P4_SUBPROGRAM  (-4)   /* a SubProgram* — trigger bodies */
#define P4_TABLE       (-5)   /* a Table* */
#define P4_DYNAMIC     (-6)   /* memory from sqliteMalloc() — freed with the program */
#define P4_FUNCDEF     (-7)
#define P4_KEYINFO     (-8)   /* collations + DESC flags for index comparisons */
#define P4_EXPR        (-9)
#define P4_MEM        (-10)
#define P4_VTAB       (-11)
#define P4_REAL       (-12)
#define P4_INT64      (-13)
#define P4_INTARRAY   (-14)
#define P4_FUNCCTX    (-15)
#define P4_TABLEREF   (-16)   /* like P4_TABLE, reference counted */
#define P4_SUBRTNSIG  (-17)
```

`P4_DYNAMIC` means the program owns the pointer and frees it; `P4_STATIC` means it does
not. Getting this wrong is a leak or a double free.

### 2.3 `OPFLG_*` — opcode properties, generated from the source comments

```c
#define OPFLG_JUMP    0x01   /* jump:   P2 holds a jump target */
#define OPFLG_IN1     0x02   /* in1:    P1 is an input register */
#define OPFLG_IN2     0x04   /* in2:    P2 is an input register */
#define OPFLG_IN3     0x08   /* in3:    P3 is an input register */
#define OPFLG_OUT2    0x10   /* out2:   P2 is an output register */
#define OPFLG_OUT3    0x20   /* out3:   P3 is an output register */
#define OPFLG_NCYCLE  0x40   /* ncycle: cycles count against P1 */
#define OPFLG_JUMP0   0x80   /* jump0:  P2 might be zero */
```

These come from the annotation after each `case` label:

```c
case OP_Integer: {         /* out2 */
case OP_Next:              /* jump, ncycle */
case OP_Eq:                /* same as TK_EQ, jump, in1, in3 */
```

`mkopcodeh.tcl` parses those comments into `sqlite3OpcodeProperty[]`. In a `SQLITE_DEBUG`
build the interpreter asserts, before every instruction, that the registers named by the
flags really are valid registers. **Mislabel an opcode and a debug build stops on the first
execution** — that is your safety net when adding one.

### 2.4 `OPFLAG_*` — P5 flags (context-dependent; the same bit means different things)

```c
#define OPFLAG_NCHANGE       0x01  /* OP_Insert/Delete: count in db->nChange */
#define OPFLAG_NOCHNG        0x01  /* OP_VColumn: no-change for UPDATE */
#define OPFLAG_EPHEM         0x01  /* OP_Column: an ephemeral result is acceptable */
#define OPFLAG_ISUPDATE      0x04  /* this OP_Insert is part of an UPDATE */
#define OPFLAG_APPEND        0x08  /* HINT: likely an append → balance_quick */
#define OPFLAG_USESEEKRESULT 0x10  /* the cursor is already positioned; skip the seek */
#define OPFLAG_LASTROWID     0x20  /* update last_insert_rowid() */
#define OPFLAG_ISNOOP        0x40  /* OP_Delete: fire the pre-update hook only */
#define OPFLAG_LENGTHARG     0x40  /* OP_Column: only length() needs this */
#define OPFLAG_TYPEOFARG     0x80  /* OP_Column: only typeof() needs this */
#define OPFLAG_BYTELENARG    0xc0  /* OP_Column: only octet_length() needs this */
#define OPFLAG_BULKCSR       0x01  /* OP_Open*: a bulk-load cursor */
#define OPFLAG_SEEKEQ        0x02  /* OP_Open*: equality seeks only */
#define OPFLAG_FORDELETE     0x08  /* OP_Open*: use BTREE_FORDELETE */
#define OPFLAG_P2ISREG       0x10  /* OP_Open*: P2 is a register, not a root page */
#define OPFLAG_PERMUTE       0x01  /* OP_Compare: use the permutation from OP_Permutation */
#define OPFLAG_SAVEPOSITION  0x02  /* OP_Delete/Insert: keep the cursor positioned */
#define OPFLAG_AUXDELETE     0x04  /* OP_Delete: an index entry, not the primary row */
#define OPFLAG_PREFORMAT     0x80  /* OP_Insert: the cell is pre-formatted */
```

`OPFLAG_TYPEOFARG` / `LENGTHARG` are a neat optimization: if the only consumer of a column
is `typeof()` or `length()`, `OP_Column` can avoid materializing the value at all.

### 2.5 `Mem` — a register / `sqlite3_value`

```c
struct sqlite3_value {
  union MemValue {
    double r;
    i64 i;
    int nZero;            /* extra zero bytes, when MEM_Zero */
    const char *zPType;
    FuncDef *pDef;
  } u;
  u16 flags;              /* MEM_* */
  u8  enc;                /* SQLITE_UTF8 / UTF16LE / UTF16BE */
  u8  eSubtype;
  int n;                  /* byte length of a string/blob */
  char *z;                /* string/blob bytes */
  char *zMalloc;          /* the allocation backing z, if owned */
  int szMalloc;
  u32 uTemp;
  sqlite3 *db;
  void (*xDel)(void*);
};
```

```c
/* type bits */
#define MEM_Undefined 0x0000  /* the value is undefined */
#define MEM_Null      0x0001
#define MEM_Str       0x0002
#define MEM_Int       0x0004
#define MEM_Real      0x0008
#define MEM_Blob      0x0010
#define MEM_IntReal   0x0020  /* an integer that stringifies like a real */
#define MEM_AffMask   0x003f
/* provenance / shape */
#define MEM_FromBind  0x0040  /* came from sqlite3_bind() */
#define MEM_Cleared   0x0100  /* NULL set by OP_Null, not read from data */
#define MEM_Term      0x0200  /* z is NUL-terminated */
#define MEM_Zero      0x0400  /* u.nZero trailing zero bytes (zeroblob) */
#define MEM_Subtype   0x0800
#define MEM_TypeMask  0x0dbf
/* OWNERSHIP of z — the field that causes most VDBE bugs */
#define MEM_Dyn       0x1000  /* call Mem.xDel() on z */
#define MEM_Static    0x2000  /* z points to static storage */
#define MEM_Ephem     0x4000  /* z is BORROWED — must be copied to survive */
#define MEM_Agg       0x8000  /* z is an aggregate-function context */
```

| Fact | Consequence |
|---|---|
| a `Mem` may have **both** `MEM_Int` and `MEM_Str` | flags are a *cache*, not just a type tag: `sqlite3_column_text()` after `_int()` is free |
| `MEM_Ephem` | the bytes belong to a page buffer or another `Mem`; `sqlite3VdbeMemMakeWriteable()` must be called before they outlive the source |
| `MEM_Zero` | `zeroblob(1000000)` costs 8 bytes in RAM, not a megabyte |
| `MEM_Undefined == 0` | a zeroed `Mem` is "undefined", which debug builds detect |

### 2.6 `Vdbe`

```c
struct Vdbe {
  sqlite3 *db;
  Vdbe *pPrev, *pNext;
  Parse *pParse;
  ynVar nVar;               /* number of parameters */
  int nMem;                 /* number of registers */
  int nCursor;
  u32 cacheCtr;             /* incremented by OP_ResultRow; invalidates column caches */
  int pc;                   /* the program counter */
  int rc;
  i64 nChange;
  Mem *aMem;                /* registers (1-based) */
  Mem **apArg;
  Mem *aColName, *pResultRow;   /* pResultRow POINTS INTO aMem — no copy */
  VdbeCursor **apCsr;
  Mem *aVar; char **azVar;
  VdbeOp *aOp; int nOp;
  u32 expmask;              /* which bound parameters force a re-prepare */
  u8 errorAction, minWriteFileFormat, prepFlags, eVdbeState;
  bft expired:2, explain:2, changeCntOn:1, usesStmtJournal:1,
      readOnly:1, bIsReader:1, runOnlyOnce:1;
  VdbeFrame *pFrame, *pDelFrame;
  SubProgram *pProgram;
  u32 aCounter[...];        /* SQLITE_STMTSTATUS_* */
};
```

### 2.7 `VdbeCursor`

```c
struct VdbeCursor {
  u8 eCurType;          /* CURTYPE_BTREE / SORTER / VTAB / PSEUDO */
  i8 iDb;
  u8 nullRow;           /* true ⇒ every OP_Column returns NULL (LEFT JOIN) */
  u8 deferredMoveto;    /* a DeferredSeek is pending */
  u8 isTable;
  Bool isEphemeral:1, useRandomRowid:1, isOrdered:1;
  Btree *pBtx;
  i16 seekHit;
  union { BtCursor *pCursor; sqlite3_vtab_cursor *pVCur; VdbeSorter *pSorter; } uc;
  KeyInfo *pKeyInfo;
  u32 cacheStatus;      /* == Vdbe.cacheCtr ⇒ aType/aOffset are VALID */
  i64 seqCount;
  i64 movetoTarget;     /* the rowid for a deferred seek */
  const u8 *aRow;       /* a pointer into the page, when the row is entirely local */
  u32 payloadSize, szRow;
  u32 *aOffset;         /* cached byte offsets of each column in the record */
  u32 aType[1];         /* cached serial types — FLEXIBLE ARRAY */
};
```

`cacheStatus` vs `Vdbe.cacheCtr` is the per-row column cache:
```
cursor moves        → pC->cacheStatus = CACHE_STALE
OP_ResultRow        → p->cacheCtr = (p->cacheCtr + 2)|1
OP_Column           → if( pC->cacheStatus!=p->cacheCtr ) re-parse the record header
```
**Forgetting `CACHE_STALE` after moving a cursor returns stale column values with no
error.** It is the classic VDBE patch bug.

### 2.8 `VdbeFrame` — the trigger / subprogram stack

```c
struct VdbeFrame {
  Vdbe *v;
  VdbeFrame *pParent;
  Mem *aMem; VdbeCursor **apCsr; VdbeOp *aOp;
  int pc, nOp, nMem, nCursor, nChildMem, nChildCsr;
  i64 nChange;
  ...
};
```
`OP_Program` pushes one; `OP_Param` reads `OLD.`/`NEW.` values out of the parent frame.

### 3.1 The twenty opcodes that appear in almost every program

```
Init       P1 P2 * P4 *      jump to P2 (the prologue trick); P4 = the SQL text
Goto       * P2 * * *        pc = P2
Gosub      P1 P2 * * *       r[P1] = return address; pc = P2
Return     P1 * * * *        pc = r[P1]
Halt       P1 P2 * P4 P5     stop; P1 = result code, P2 = ON CONFLICT action
Transaction P1 P2 P3 P4 P5   start a read (P2=0) / write (P2=1) txn on db P1;
                             P3 = expected schema cookie → SQLITE_SCHEMA if stale
OpenRead   P1 P2 P3 P4 P5    cursor P1 ← root page P2 of db P3; P4 = KeyInfo or nCol
OpenWrite  ...               same, for writing
Rewind     P1 P2 * * *       first entry of cursor P1; jump P2 if empty
Next       P1 P2 P3 P4 P5    advance cursor P1; jump back to P2 if a row exists
Column     P1 P2 P3 P4 P5    r[P3] = column P2 of the row at cursor P1
Rowid      P1 P2 * * *       r[P2] = the rowid at cursor P1
SeekRowid  P1 P2 P3 * *      position cursor P1 at rowid r[P3]; jump P2 if absent
SeekGE/GT/LE/LT P1 P2 P3 P4  seek index cursor P1 using P4 values from r[P3]
IdxGT/GE/LT/LE  P1 P2 P3 P4  compare the current index key with r[P3..]; jump
IdxRowid   P1 P2 * * *       r[P2] = the rowid stored in the index entry
IdxInsert  P1 P2 P3 P4 P5    insert record r[P2] into index cursor P1
MakeRecord P1 P2 P3 P4 P5    r[P3] = record from r[P1..P1+P2-1]; P4 = affinity string
Insert     P1 P2 P3 P4 P5    write record r[P2] with rowid r[P3] into cursor P1
ResultRow  P1 P2 * * *       return r[P1..P1+P2-1]; sqlite3_step → SQLITE_ROW
```

## 4. Program-building API (`vdbeaux.c`)

```c
int  sqlite3VdbeAddOp0/1/2/3(Vdbe*, int op, ...);            /* returns the ADDRESS */

int  sqlite3VdbeMakeLabel(Parse*);              /* returns a NEGATIVE pseudo-address */
void sqlite3VdbeResolveLabel(Vdbe*, int x);
void sqlite3VdbeJumpHere(Vdbe*, int addr);      /* set aOp[addr].p2 = current address */
void sqlite3VdbeJumpHereOrPopInst(Vdbe*, int addr);
void sqlite3VdbeChangeP1/P2/P3/P5(Vdbe*, int addr, int val);
void sqlite3VdbeChangeP4(Vdbe*, int addr, const char *z, int n);
void sqlite3VdbeAppendP4(Vdbe*, void *p, int p4type);
int  sqlite3VdbeCurrentAddr(Vdbe*);
void sqlite3VdbeComment(Vdbe*, const char*, ...);      /* VdbeComment((v, ...)) */
void sqlite3VdbeVerifyAbortable(Vdbe*, int);
```

Two forward-jump idioms:

```c
/* patch by address */
int a = sqlite3VdbeAddOp2(v, OP_If, reg, 0);
... body ...
sqlite3VdbeJumpHere(v, a);

/* labels, when the target is used more than once */
int lbl = sqlite3VdbeMakeLabel(pParse);
sqlite3VdbeAddOp2(v, OP_Goto, 0, lbl);
...
sqlite3VdbeResolveLabel(v, lbl);
```

`resolveP2Values()` runs a final pass that binds labels **and** computes `p->readOnly`,
`p->bIsReader`, and whether a statement journal is required.

## 5. Expression codegen (`expr.c`)

| Function | Produces |
|---|---|
| `sqlite3ExprCode(pParse, pExpr, target)` | the value in register `target` |
| `sqlite3ExprCodeTarget(pParse, pExpr, target)` | the value; **returns the register actually used** (may avoid a copy) |
| `sqlite3ExprCodeTemp(pParse, pExpr, &reg)` | the value in some register |
| `sqlite3ExprIfTrue(pParse, pExpr, dest, jumpIfNull)` | **jumps** to `dest` if true — no boolean materialized |
| `sqlite3ExprIfFalse(...)` | jumps if false |
| `sqlite3ExprCodeExprList(...)` | a contiguous run of registers |
| `sqlite3ExprCodeRunJustOnce(...)` | hoists a constant into the prologue |
| `sqlite3ExprCodeMove(...)` | `OP_Move` a range |

Register allocation:
```c
int target = ++pParse->nMem;               /* permanent */
int reg    = sqlite3GetTempReg(pParse);    /* from an 8-slot pool */
sqlite3ReleaseTempReg(pParse, reg);
int base   = sqlite3GetTempRange(pParse, n);
sqlite3ReleaseTempRange(pParse, base, n);
```

`jumpIfNull` (`SQLITE_JUMPIFNULL`) threads SQL's three-valued logic through the
jump-based transformation. Getting it wrong changes the meaning of `NOT (x = NULL)`.

## 6. `SelectDest.eDest` — one query body, twelve consumers

| Constant | Rows go to |
|---|---|
| `SRT_Output` | `OP_ResultRow` → the application |
| `SRT_Mem` | a single register (a scalar subquery) |
| `SRT_Set` | an ephemeral index (for `IN (...)`) |
| `SRT_Table` / `SRT_EphemTab` | an ephemeral table |
| `SRT_Coroutine` | yielded to a consumer, row by row |
| `SRT_Exists` | set a register to 1 and stop |
| `SRT_Union` / `SRT_Except` | compound-select accumulation |
| `SRT_Fifo` / `SRT_DistFifo` / `SRT_Queue` / `SRT_DistQueue` | recursive CTE queues |

| `SRT_Upfrom` | `UPDATE ... FROM` |

Reading `selectInnerLoop()`'s `switch(eDest)` is the fastest way to see how one
implementation serves every query context.

## 7. Statement lifecycle and status

```
sqlite3_prepare_v2 → parse + plan + codegen → sqlite3VdbeMakeReady()
                     (aMem, apCsr, aVar allocated in ONE block)
sqlite3_bind_*     → write p->aVar[i]
sqlite3_step       → sqlite3Step → sqlite3VdbeExec; SQLITE_ROW at OP_ResultRow
sqlite3_column_*   → read p->pResultRow[i]   (NO COPY)
sqlite3_reset      → rewind pc, close cursors, KEEP bindings
sqlite3_finalize   → free everything
```

```c
sqlite3_stmt_status(pStmt, SQLITE_STMTSTATUS_FULLSCAN_STEP, reset);
                            SQLITE_STMTSTATUS_SORT
                            SQLITE_STMTSTATUS_AUTOINDEX
                            SQLITE_STMTSTATUS_VM_STEP
                            SQLITE_STMTSTATUS_REPREPARE
                            SQLITE_STMTSTATUS_RUN
                            SQLITE_STMTSTATUS_FILTER_HIT / _FILTER_MISS
                            SQLITE_STMTSTATUS_MEMUSED
```
`FULLSCAN_STEP` is incremented by `OP_Next` when P5 says so — that is how you detect a
full scan at runtime without parsing `EXPLAIN`.

## 8. Invariants

1. Registers are **1-based**; `aMem[0]` is unused.
2. An opcode that fails must `goto abort_due_to_error` — never fall through with `rc` set.
3. `pC->cacheStatus = CACHE_STALE` after **every** cursor movement.
4. `p->pResultRow` points into `aMem`; those registers must not be reused before the next
   `step()`.
5. `OP_ResultRow` saves `pc` and returns; the VM resumes at the saved address.
6. `/* inN outN jump */` annotations must match the actual operand usage, or a debug build
   asserts.
7. `sqlite3VdbeAddOp*` returns the instruction **address**; labels are negative until
   resolved.
8. `MEM_Ephem` values must be made writeable before they outlive the cursor position.
9. `sqlite3VdbeMakeReady()` is called exactly once per statement.
10. A statement holds a read transaction from the first `step()` until `reset`/`finalize`.

---

## 9. Gotchas

1. **Opcode numbers change between releases.** Use names; never persist a number.
2. `EXPLAIN` and `EXPLAIN QUERY PLAN` do **not** execute the statement.
3. `sqlite3_column_*` pointers are invalidated by the next `step()`, `reset()`, or
   `finalize()`.
4. `sqlite3_column_text()` may **convert the register in place**, invalidating a pointer
   previously returned by `sqlite3_column_blob()`.
5. Reading a later column costs more — `OP_Column` walks the record header.
6. `ALTER TABLE ADD COLUMN` is O(1) because `OP_Column` returns the P4 default when the
   stored record is short.
7. `OP_Once` is reset by `sqlite3_reset()`; a subprogram's `Once` needs care.
8. `P4_DYNAMIC` transfers ownership to the program; `P4_STATIC` does not.
9. `OPFLAG_USESEEKRESULT` asserts that the cursor really is positioned — a stale value
   silently corrupts.
10. `sqlite3_expanded_sql()` is a **reconstruction**, not the executed text.
11. An unfinalized statement holds a read transaction — the top cause of `SQLITE_BUSY`
    and of unbounded WAL growth.
12. Trigger recursion is bounded by `SQLITE_MAX_TRIGGER_DEPTH`; unbounded recursion is a
    resource-exhaustion bug class.

## 10. Debug recipes

```sql
PRAGMA vdbe_listing=ON;     -- print the program before running
PRAGMA vdbe_trace=ON;       -- print every instruction as it executes
PRAGMA vdbe_addoptrace=ON;
.scanstats on               -- per-loop estimated vs actual rows
EXPLAIN SELECT ...;
SELECT addr,opcode,p1,p2,p3,p4,p5,comment FROM bytecode('SELECT ...');
SELECT * FROM tables_used('SELECT ...');
```

Analysis queries worth keeping:

```sql
-- does this query sort?
SELECT count(*)>0 FROM bytecode('...') WHERE opcode LIKE 'Sorter%';

-- which cursors are opened, and on what?
SELECT addr,opcode,p1 AS cursor,p2 AS rootpage,p4
  FROM bytecode('...') WHERE opcode LIKE 'Open%';

-- opcode histogram
SELECT opcode,count(*) FROM bytecode('...') GROUP BY opcode ORDER BY 2 DESC;
```

```
(lldb) b sqlite3VdbeExec
(lldb) p p->nOp
(lldb) p p->nMem
(lldb) call (void)sqlite3VdbePrintOp(0, 0, pOp)
(lldb) p p->aMem[1]
(lldb) p/x p->aMem[1].flags        # MEM_*
(lldb) p p->apCsr[0]->uc.pCursor->pgnoRoot
(lldb) p p->apCsr[0]->cacheStatus
(lldb) b vdbe.c:<line of case OP_Column>
(lldb) breakpoint modify -c "pOp->p2 == 3"
```

## 11. Adding an opcode — the checklist

1. Add `case OP_Xxx:` in `vdbe.c` with a `/* Opcode: ... */` doc comment **above** it and
   the correct `/* out2 */`-style annotation **after** the label.
2. `make opcodes.h opcodes.c` and confirm `OP_Xxx` appears.
3. Emit it from codegen (`expr.c`, `select.c`, or wherever it belongs).
4. Use `out2Prerelease()` for an out2 opcode; `sqlite3VdbeMemSetStr()` for strings.
5. Build with `SQLITE_DEBUG` — the annotation asserts will catch operand mistakes.
6. Run `./testfixture ../test/select1.test` and `../test/func.test`.
7. Never hard-code the numeric value anywhere.
